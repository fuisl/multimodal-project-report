\paragraph{Retrieval Quality Framework (for RAG)}
Evaluates the effectiveness of knowledge retrieval:
\begin{itemize}
    \item \textbf{Retrieval Accuracy:} Relevance of retrieved documents
    \item \textbf{Context Relevance:} Usefulness of retrieved knowledge for diagnosis

\paragraph{Safety and Reliability Framework}
Assesses system reliability for clinical deployment:
\begin{itemize}
    \item \textbf{Calibration:} Alignment between confidence and accuracy
    \item \textbf{Uncertainty Quantification:} Identification of uncertain cases
    \item \textbf{Failure Analysis:} Understanding error patterns
    \item \textbf{Robustness:} Performance under input variations
\end{itemize}


\subsubsection{Evaluation Metrics}
\label{sec:eval-metrics}

\paragraph{Primary Diagnostic Metrics}
Core metrics for diagnostic performance assessment:

\begin{enumerate}
    \item \textbf{Accuracy:} Overall correctness of predictions
    \begin{equation}
        \text{Accuracy} = \frac{\text{Correct Predictions}}{\text{Total Predictions}}
    \end{equation}
    
    \item \textbf{Precision (Positive Predictive Value):}
    \begin{equation}
        \text{Precision} = \frac{\text{True Positives}}{\text{True Positives} + \text{False Positives}}
    \end{equation}
    
    \item \textbf{Recall (Sensitivity):}
    \begin{equation}
        \text{Recall} = \frac{\text{True Positives}}{\text{True Positives} + \text{False Negatives}}
    \end{equation}
    
    \item \textbf{F1-Score:} Harmonic mean of precision and recall
    \begin{equation}
        \text{F1} = 2 \times \frac{\text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}}
    \end{equation}
    
    \item \textbf{Specificity:} True negative rate
    \begin{equation}
        \text{Specificity} = \frac{\text{True Negatives}}{\text{True Negatives} + \text{False Positives}}
    \end{equation}
\end{enumerate}

\paragraph{Multi-class and Multi-label Metrics}
For handling multiple diseases and differential diagnoses:

\begin{itemize}
    \item \textbf{Macro-averaged F1:} Unweighted mean of per-class F1 scores (treats all diseases equally)
    \item \textbf{Micro-averaged F1:} Aggregate contributions of all classes (favors frequent diseases)
    \item \textbf{Weighted F1:} F1 weighted by class frequency
    \item \textbf{Top-K Accuracy:} Percentage of cases where correct diagnosis is in top K predictions
    \item \textbf{Mean Reciprocal Rank (MRR):} Average of reciprocal ranks of correct diagnoses
\end{itemize}

\paragraph{Clinical Utility Metrics}
Metrics tailored for clinical decision support:

\begin{itemize}
    \item \textbf{AUROC (Area Under ROC Curve):} Discriminative ability across thresholds
    \item \textbf{AUPRC (Area Under Precision-Recall Curve):} Performance on imbalanced data
    \item \textbf{Cohen's Kappa:} Agreement beyond chance
    \item \textbf{Matthews Correlation Coefficient (MCC):} Balanced measure for imbalanced datasets
    \item \textbf{Diagnostic Odds Ratio (DOR):} Effectiveness of diagnostic test
\end{itemize}

\paragraph{RAG-specific Metrics}
For evaluating retrieval-augmented systems:

\begin{itemize}
    \item \textbf{Retrieval Precision@K:} Proportion of relevant documents in top K
    \item \textbf{Mean Average Precision (MAP):} Average precision across queries
    \item \textbf{Normalized Discounted Cumulative Gain (NDCG):} Ranking quality
    \item \textbf{Context Relevance Score:} Relevance of retrieved context to diagnosis
    \item \textbf{Faithfulness:} Consistency between generated answer and retrieved context
    \item \textbf{Answer Relevance:} Relevance of generated answer to query
\end{itemize}

\paragraph{Fine-tuning Specific Metrics}
Additional metrics for trained models:

\begin{itemize}
    \item \textbf{Perplexity:} Language model quality
    \item \textbf{BLEU/ROUGE:} Quality of generated diagnostic explanations (if applicable)
    \item \textbf{Training Loss Convergence:} Model learning effectiveness
    \item \textbf{Validation Loss:} Generalization capability
\end{itemize}

\paragraph{Hybrid/Combination Metrics}
For combined approaches:

\begin{itemize}
    \item \textbf{Ensemble Agreement:} Consistency between RAG and fine-tuned predictions
    \item \textbf{Complementarity Score:} How well methods complement each other
    \item \textbf{Fusion Effectiveness:} Performance gain from combination
\end{itemize}

\paragraph{Interpretability Metrics}
Assessing model explainability:

\begin{itemize}
    \item \textbf{Reasoning Quality Score:} Expert assessment of diagnostic reasoning (1-5 scale)
    \item \textbf{Evidence Grounding:} Percentage of claims supported by retrieved evidence
    \item \textbf{Clinical Coherence:} Alignment with clinical reasoning standards
\end{itemize}


\subsubsection{Evaluation Protocol}
\label{sec:eval-protocol}

\paragraph{Test Set Evaluation}
\begin{enumerate}
    \item \textbf{Inference:} Generate predictions for all test cases
    \item \textbf{Metric Computation:} Calculate all relevant metrics
    \item \textbf{Stratified Analysis:} Performance breakdown by:
    \begin{itemize}
        \item Disease category (e.g., bacterial, viral, parasitic)
        \item Disease prevalence (common vs. rare)
        \item Case complexity (simple vs. multi-system)
        \item Data modality (text-only vs. multimodal)
    \end{itemize}
\end{enumerate}

\paragraph{Cross-validation (if applicable)}
\begin{itemize}
    \item Strategy: [e.g., 5-fold cross-validation]
    \item Purpose: Assess model stability and reduce evaluation variance
    \item Report: Mean and standard deviation across folds
\end{itemize}

\paragraph{Baseline Comparisons}
Compare against established baselines:
\begin{itemize}
    \item \textbf{Random Baseline:} Random diagnosis selection
    \item \textbf{Majority Class Baseline:} Always predict most frequent disease
    \item \textbf{Traditional ML:} [e.g., Random Forest, XGBoost on extracted features]
    \item \textbf{Existing Models:} [e.g., Clinical BERT, BioGPT]
    \item \textbf{Human Performance:} [If available - clinician diagnostic accuracy]
\end{itemize}

\paragraph{Ablation Studies}
Systematic evaluation of component contributions:
\begin{itemize}
    \item \textbf{Modality Ablation:} Text-only vs. image-only vs. multimodal
    \item \textbf{RAG Ablation:} With vs. without retrieval
    \item \textbf{Fine-tuning Ablation:} Base model vs. fine-tuned
    \item \textbf{Component Ablation:} Impact of specific architectural components
\end{itemize}


\subsubsection{Statistical Significance Testing}
\begin{itemize}
    \item \textbf{McNemar's Test:} Pairwise comparison of model predictions
    \item \textbf{Bootstrap Confidence Intervals:} Estimate metric uncertainty
    \item \textbf{Significance Level:} $\alpha = 0.05$ (p-value threshold)
\end{itemize}


\subsubsection{Error Analysis Framework}
Systematic investigation of model failures:
\begin{enumerate}
    \item \textbf{Error Categorization:}
    \begin{itemize}
        \item Misdiagnosis: Incorrect disease identified
        \item Missed diagnosis: Failed to identify disease
        \item Overdiagnosis: False positive
        \item Reasoning error: Correct diagnosis, flawed reasoning
    \end{itemize}
    
    \item \textbf{Root Cause Analysis:}
    \begin{itemize}
        \item Data quality issues
        \item Model limitations
        \item Knowledge gaps
        \item Input ambiguity
    \end{itemize}
    
    \item \textbf{Severity Assessment:}
    \begin{itemize}
        \item Critical: Potentially harmful misdiagnosis
        \item Major: Significant diagnostic error
        \item Minor: Acceptable alternative diagnosis
    \end{itemize}
\end{enumerate}
